void Timer(){
std::cout << "\x1B[2J\x1B[3J\x1B[H";
int sec = 0;
std::cout << "Timer\n";
std::cout << "-------------------------\n\n";
std::cout << "-------------------------\n";
std::cout << "\x1B[2AHow many seconds: ";
std::cin >> sec;
std::cout << "\x1B[2J\x1B[3J\x1B[H";
for(int i = 1;i <= sec;i++){
std::cout << "Seconds: " << i << std::flush;
std::this_thread::sleep_for(std::chrono::seconds(1));
std::cout << "\x1B[2J\x1B[3J\x1B[H";
}
std::cout << "Alert\n";
std::cout << "-------------------------\n";
Ask(Timer);
}

void Calculator(){
std::cout << "\x1B[2J\x1B[3J\x1B[H";
float a = 0.f, b = 0.f, c = 0.f;
bool isvalid = 1;
char op = '\0';
std::cout << "Calculator\n";
std::cout << "-------------------------\n\n";
std::cout << "-------------------------\n";
std::cout << "\x1B[2ANumber One: ";
std::cin >> a;
std::cout << "\x1B[2J\x1B[3J\x1B[H";
std::cout << "Calculator\n";
std::cout << "-------------------------\n\n";
std::cout << "-------------------------\n";
std::cout << "\x1B[2AEnter +, -, *, /: ";
std::cin >> op;
std::cout << "\x1B[2J\x1B[3J\x1B[H";
std::cout << "Calculator\n";
std::cout << "-------------------------\n\n";
std::cout << "-------------------------\n";
std::cout << "\x1B[2ANumber Two: ";
std::cin >> b;
std::cout << "\x1B[2J\x1B[3J\x1B[H";
switch(op){
case '+': c = a+b;break;
case '-': c = a-b;break;
case '*': c = a*b;break;
case '/': c = a/b;break;
default: isvalid = 0;break;
}
std::cout << "Result\n";
std::cout << "-------------------------\n";
if(isvalid)std::cout << a << ' ' << op << ' ' << b << " = " << c << '\n';
else std::cout << "\x1B[33mInvalid Operator!\n\x1B[0m";
std::cout << "-------------------------\n";
Ask(Calculator);
}

void NumberGuessing(){
std::cout << "\x1B[2J\x1B[3J\x1B[H";
srand(time(0));
int guess = 0, tries = 0, num = rand()%100+1;
std::cout << "Number Guessing (1 - 100)\n";
std::cout << "Enter '0' to exit.\n";
std::cout << "-------------------------\n";
while(true){
tries++;
std::cout << "Your Guess: ";
std::cin >> guess;
if(guess == num)break;
else if(guess < num && guess != 0)std::cout << "\n\x1B[31mToo low!\x1B[0m\n\n";
else if(guess > num && guess != 0)std::cout << "\n\x1B[31mToo high!\x1B[0m\n\n";
else if(guess == 0)GameSection();
}
std::cout << "\n\x1B[32mCorrect! You guessed " << num << " in " << tries << " tries.\n\x1B[0m";
std::cout << "-------------------------\n";
GameAsk(NumberGuessing);
}

void RPS(){
std::cout << "\x1B[2J\x1B[3J\x1B[H";
srand(time(0));
int ai = 0;
char pl = '\0';
std::cout << "Rock Paper Scissors!\n";
std::cout << "Rock - r, Paper - p, Scissors - s (Exit - e)\n";
std::cout << "-------------------------\n";
while(true){
ai = rand()%3;
std::cout << "Your Try: ";
std::cin >> pl;
if(pl == 'e')break;
//Tie
else if(pl == 'r' && ai == 0)std::cout << "\n\x1B[33mI have Rock too. Tie!\x1B[0m\n\n";
else if(pl == 'p' && ai == 1)std::cout << "\n\x1B[33mI have Paper too. Tie!\x1B[0m\n\n";
else if(pl == 's' && ai == 2)std::cout << "\n\x1B[33mI have Scissors too. Tie!\x1B[0m\n\n";
//Won
else if(pl == 'r' && ai == 2)std::cout << "\n\x1B[32mI have Scissors. You Won!\x1B[0m\n\n";
else if(pl == 'p' && ai == 0)std::cout << "\n\x1B[32mI have Rock. You Won!\x1B[0m\n\n";
else if(pl == 's' && ai == 1)std::cout << "\n\x1B[32mI have Paper. You Won!\x1B[0m\n\n";
//Lose
else if(pl == 'r' && ai == 1)std::cout << "\n\x1B[31mI have Paper. You Lost!\x1B[0m\n\n";
else if(pl == 'p' && ai == 2)std::cout << "\n\x1B[31mI have Scissors. You Lost!\x1B[0m\n\n";
else if(pl == 's' && ai == 0)std::cout << "\n\x1B[31mI have Rock. You Lost!\x1B[0m\n\n";
/////
else std::cout << "\n\x1B[33mInvalid Try!\x1B[0m\n\n";
}
std::cout << "\x1B[2J\x1B[3J\x1B[H";
std::cout << "-------------------------\n";
GameAsk(RPS);
}
